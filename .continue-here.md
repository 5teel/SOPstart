```ts
import type { SupabaseClient } from '@supabase/supabase-js'

const SIGNED_TTL_SEC = 3600

type SrcCarrier = { src?: unknown }
type ItemList = { items?: SrcCarrier[]; photos?: SrcCarrier[]; src?: unknown }

/**
 * Walk every section's layout_data and rewrite raw sop-images storage paths
 * to 1-hour signed URLs. Mutates `sop` in-place.
 *
 * Recognises src on:
 *   - PhotoBlock.props.src
 *   - StepWithPhotosBlock.props.photos[].src
 *   - PhotoGridBlock.props.items[].src
 */
export async function signLayoutDataImages(
  supabase: SupabaseClient,
  sop: { sop_sections?: Array<{ layout_data?: unknown }> }
): Promise<void> {
  // 1. Collect every storage-path src across all sections.
  const refs: Array<{ obj: SrcCarrier; path: string }> = []

  for (const section of sop.sop_sections ?? []) {
    const ld = section.layout_data as { content?: Array<{ type?: string; props?: ItemList }> } | null | undefined
    if (!ld?.content) continue
    for (const block of ld.content) {
      const props = block.props
      if (!props) continue
      // PhotoBlock-style single src
      if (typeof props.src === 'string' && !/^https?:/i.test(props.src) && !props.src.startsWith('data:')) {
        refs.push({ obj: props as SrcCarrier, path: props.src as string })
      }
      // StepWithPhotosBlock.photos[].src
      for (const p of props.photos ?? []) {
        if (typeof p.src === 'string' && !/^https?:/i.test(p.src) && !p.src.startsWith('data:')) {
          refs.push({ obj: p, path: p.src as string })
        }
      }
      // PhotoGridBlock.items[].src
      for (const it of props.items ?? []) {
        if (typeof it.src === 'string' && !/^https?:/i.test(it.src) && !it.src.startsWith('data:')) {
          refs.push({ obj: it, path: it.src as string })
        }
      }
    }
  }
  if (refs.length === 0) return

  // 2. Batch sign in one round-trip.
  const paths = refs.map((r) => r.path)
  const { data } = await supabase.storage
    .from('sop-images')
    .createSignedUrls(paths, SIGNED_TTL_SEC)
  if (!data) return

  // 3. Mutate in place.
  for (let i = 0; i < refs.length; i++) {
    const signed = data[i]?.signedUrl
    if (signed) (refs[i].obj as SrcCarrier).src = signed
  }
}
```
