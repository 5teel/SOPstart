  <rationale>
    **Idempotency on retry — accepted product behavior (W-3).**

    Each POST to /api/sops/ai-prompt creates a fresh `sops` row + `parse_jobs` row + Anthropic call. There is NO request-id-based idempotency guard, and no "look up an existing job for this prompt before creating one" pre-check. This differs from `transcribe/route.ts` (which guards on existing-job lookup before re-running).

    **Why accept it:**
    - The PromptClient has no notion of a request_id to send; adding one would require a client-side UUID and a server-side dedup table.
    - Duplicate submissions are easy for admins to spot in /admin/sops (two AI DRAFT chips with the same title, seconds apart) and easy to delete.
    - Real-world duplicate rate is expected to be <1% (admin clicks Generate, sees stepper render, doesn't refresh-spam during the 10-30s pipeline).
    - Cost cap: Zod max(2000) + Haiku triage + admin-only access bound the worst-case cost of one duplicate run to ~$0.02-0.05.

    **Trade-off accepted:** Simpler request-id-less semantics in exchange for a small duplication risk. **Revisit if cost telemetry shows real-world duplication exceeds ~5% of runs.**
  </rationale>

export async function POST(request: NextRequest) {
  // --- 1. Auth + role guard (copy lines 14-35 of youtube/route.ts verbatim) ---
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Not authenticated' }, { status: 401 })

  const { data: { session } } = await supabase.auth.getSession()

    ? JSON.parse(atob(session.access_token.split('.')[1]))

  const organisationId: string | null = jwtClaims['organisation_id'] ?? null

  const parseResult = aiPromptSchema.safeParse(body)
  if (!parseResult.success) {
    return NextResponse.json(
      { error: parseResult.error.issues[0]?.message ?? 'Invalid prompt' },
      { status: 400 }
    )

  const { promptText, categoryTag, detailLevel } = parseResult.data

  //   - Column is NOT NULL (00003), so we MUST send a value.
  //   - Current CHECK (00012) permits 'txt' alongside docx/pdf/image/xlsx/pptx/video.
  //   - 'txt' is the closest semantic match — the prompt IS a short text input. NO migration needed.
  const { data: sop, error: sopError } = await admin
    .from('sops')
    .insert({
      organisation_id: organisationId,
      title: null,
      status: 'parsing',
      version: 1,
      source_file_path: `ai-prompt/${user.id}/${Date.now()}`,
      source_file_type: 'txt',
      source_file_name: 'AI prompt',
      source_type: 'ai',         // D-05 — drives the 'AI DRAFT' library chip in 14-03
      category: categoryTag ?? null,
      is_ocr: false,
      uploaded_by: uploadedBy,
    })
    .select('id')
    .single()
  if (sopError || !sop) return NextResponse.json({ error: 'Failed to create SOP record' }, { status: 500 })

  // --- 4. Create parse_jobs row (D-04 persists prompt_text for audit) ---
  const { data: job, error: jobError } = await admin
    .from('parse_jobs')
    .insert({
      organisation_id: organisationId,
      sop_id: sop.id,
      status: 'processing',
      file_path: `ai-prompt/${user.id}`,
      file_type: 'txt',
      input_type: 'ai_prompt',     // 14-01 migration 00029 extended CHECK to permit this
      current_stage: 'prompting',  // first stage of STAGE_SETS.ai_prompt
      prompt_text: promptText,     // D-04 — audit trail
      started_at: new Date().toISOString(),
    })
    .select('id, status')
    .single()
  if (jobError || !job) return NextResponse.json({ error: 'Failed to create parse job' }, { status: 500 })

  // --- (No idempotency guard — see Task 2 rationale block above. Each POST is a fresh run.) ---

  try {
    // --- 5. Drafting stage — call Claude via the extended parseSopWithGPT signature ---
    await admin.from('parse_jobs')
      .update({ current_stage: 'drafting', updated_at: new Date().toISOString() })
      .eq('id', job.id)

    const parsed: ParsedSop = await parseSopWithGPT(promptText, { sourceMode: 'prompt', detailLevel })

    // --- 6. Verifying stage — adversarial verifier in PROMPT MODE (D-02; param implemented in 14-03) ---
    await admin.from('parse_jobs')
      .update({ current_stage: 'verifying', updated_at: new Date().toISOString() })
      .eq('id', job.id)

    const adversarialFlags = await verifyTranscriptVsSop(promptText, parsed, { mode: 'prompt' })
    const missingSectionFlags = detectMissingSections(parsed)
    const allFlags: VerificationFlag[] = [...adversarialFlags, ...missingSectionFlags]

    await admin.from('parse_jobs')
      .update({ verification_flags: allFlags as unknown as import('@/types/database.types').Json })
      .eq('id', job.id)

    // --- 7. Resolve section_kind_id for each parsed section (ROADMAP success #4) ---
    // One-shot fetch of canonical kinds; build slug -> id map for substring-match lookup.
    const { data: kindRows } = await admin.from('section_kinds').select('id, slug')
    const slugToKindId = new Map<string, string>()
    for (const k of kindRows ?? []) {
      slugToKindId.set(k.slug.toLowerCase(), k.id)
    }
    function resolveKindId(sectionType: string): string | null {
      const t = sectionType.toLowerCase()
      // Exact match first
      if (slugToKindId.has(t)) return slugToKindId.get(t)!
      // Substring match against canonical slugs (legacy renderer fallback shape)
      for (const [slug, id] of slugToKindId) {
        if (t.includes(slug)) return id
      }
      return null
    }

    // --- 8. Persist parsed SOP fields ---
    await admin.from('sops')
      .update({
        title: parsed.title,
        sop_number: parsed.sop_number ?? null,
        revision_date: parsed.revision_date ?? null,
        author: parsed.author ?? null,
        category: parsed.category ?? categoryTag ?? null,
        related_sops: parsed.related_sops ?? null,
        applicable_equipment: parsed.applicable_equipment ?? null,
        required_certifications: parsed.required_certifications ?? null,
        overall_confidence: parsed.overall_confidence,
        parse_notes: parsed.parse_notes ?? null,
        status: 'draft',
        updated_at: new Date().toISOString(),
      })
      .eq('id', sop.id)

    // --- 9. Persist sections + steps (with section_kind_id resolved) ---
    // W-5 confirmation: ParsedSopSchema (validators/sop.ts) defines
    //   SopSectionSchema.order  -> mapped to sop_sections.sort_order
    //   SopStepSchema.order     -> mapped to sop_steps.step_number
    // Verified via direct read of validators/sop.ts in this session.
    for (const section of parsed.sections) {
      const sectionKindId = resolveKindId(section.type)
      const { data: sectionRow, error: sectionError } = await admin
        .from('sop_sections')
        .insert({
          sop_id: sop.id,
          section_type: section.type,
          section_kind_id: sectionKindId,  // ROADMAP success #4 — this is the new resolver hook
          title: section.title,
          content: section.content ?? null,
          sort_order: section.order,
          confidence: section.confidence,
          approved: false,
        })
        .select('id')
        .single()
      if (sectionError || !sectionRow) continue

      if (section.steps?.length) {
        for (const step of section.steps) {
          await admin.from('sop_steps').insert({
            section_id: sectionRow.id,
            step_number: step.order,
            text: step.text,
            warning: step.warning ?? null,
            caution: step.caution ?? null,
            tip: step.tip ?? null,
            required_tools: step.required_tools ?? null,
            time_estimate_minutes: step.time_estimate_minutes ?? null,
          })
        }
      }
    }

    // --- 10. Mark job completed ---
    await admin.from('parse_jobs')
      .update({
        status: 'completed',
        current_stage: 'completed',
        completed_at: new Date().toISOString(),
      })
      .eq('id', job.id)

    return NextResponse.json({ success: true, sopId: sop.id })
  } catch (error) {
    const message = error instanceof Error ? error.message : 'Unknown error'
    console.error('AI prompt pipeline error:', message)

    const { data: currentJob } = await admin
      .from('parse_jobs')
      .select('current_stage')
      .eq('id', job.id)
      .single()

    await admin.from('parse_jobs')
      .update({
        status: 'failed',
        current_stage: 'failed',
        error_message: `Failed at ${currentJob?.current_stage ?? 'unknown'}: ${message}`,
        retry_count: 1,
      })
      .eq('id', job.id)

    return NextResponse.json({ error: message }, { status: 500 })
  }
}
